ad_name VARCHAR(50) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE media (
media_id   INT PRIMARY KEY,
media_name VARCHAR(50) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE ad_post (
post_id     INT PRIMARY KEY,
ad_id       INT NOT NULL,
media_id    INT NOT NULL,
start_dt    DATETIME NOT NULL,

) ENGINE=InnoDB;

-- 3) "매체ID별 최초 게시 광고" + "광고시작일자" 조회 (문제 25번 형태와 동일한 로직)
SELECT
m.media_name,
a.ad_name,
p.start_dt
FROM ad_post p
JOIN ad a ON p.ad_id = a.ad_id
JOIN media m ON p.media_id = m.media_id
JOIN (
SELECT media_id, MIN(start_dt) AS start_dt
FROM ad_post
GROUP BY media_id
) d
ON p.media_id = d.media_id
AND p.start_dt = d.start_dt
ORDER BY m.media_name;
